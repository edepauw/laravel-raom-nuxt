title: Paginated list
description: A page-by-page list with prev/next controls, driven by the pagination tuple.

---

`.getPage(n)` returns the `[collection, pagination]` tuple. Keep the current page in a ref and refetch when it changes.

```vue [pages/reports/index.vue]
<script setup lang="ts">
import type { ModelCollection } from 'laravel-raom-nuxt/runtime'

const reports = ref<ModelCollection<Report>>([])
const page = ref(1)
const lastPage = ref(1)
const total = ref(0)

async function load() {
  const [data, pagination] = await Report.query()
    .include('creator')
    .orderBy('created_at', 'desc')
    .limit(10)
    .getPage(page.value)

  reports.value = data
  lastPage.value = pagination.last_page
  total.value = pagination.total
}

await load()          // first page, fetched during SSR
watch(page, load)     // refetch whenever the page changes
</script>

<template>
  <div>
    <ul>
      <li
        v-for="report in reports"
        :key="report.id"
      >
        {{ report.title }} — {{ report.creator?.name }}
      </li>
    </ul>

    <nav class="flex items-center gap-3">
      <button
        :disabled="page <= 1"
        @click="page--"
      >
        Previous
      </button>

      <span>Page {{ page }} / {{ lastPage }} · {{ total }} total</span>

      <button
        :disabled="page >= lastPage"
        @click="page++"
      >
        Next
      </button>
    </nav>
  </div>
</template>
```

::warning
Type the ref as `ModelCollection<Report>`, never plain `Report[]`. The value is a raw array during SSR and a `ModelList<Report>` once hydrated — `ModelCollection<T>` is the interface that unifies both. See [Lists, SSR & identity](/guide/lists-ssr-and-identity).
::

::tip
Use `pagination.total` (from the tuple) for "items returned by the API", not `reports.length` — a `ModelList` hides instances deleted client-side, so its length is the *visible* count, not the server total.
::
